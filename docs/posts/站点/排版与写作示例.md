```ts
// TypeScript 示例：读取文章并统计字数
export function countWords(md: string): number {
  const cjk = md.match(/[一-鿿]/g)?.length ?? 0
  const latin = md.match(/[A-Za-z]+/g)?.length ?? 0
  return cjk + latin
}
```
